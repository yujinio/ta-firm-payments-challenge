+@router.get("/healthz")
+def healthz(
+    response: Response, session: Annotated[Session, Depends(get_session)]
+) -> dict[str, str]:
+    try:
+        session.execute(text("SELECT 1"))
+    except SQLAlchemyError:
+        response.status_code = 503
+        return {"status": "unavailable"}
+    return {"status": "ok"}

+@router.post(BULK_PAYMENTS_PATH, status_code=201)
+def create_bulk_payment(
+    payload: BulkPaymentRequest,
+    request: Request,
+    session: Annotated[Session, Depends(get_session)],
+    settings: Annotated[Settings, Depends(get_settings)],
+) -> JSONResponse:
+    # Fields for the per-request log record written in app.main.
+    request.state.payer = str(payload.payer_firm_uuid)
+    request.state.payment_count = len(payload.payments)
+    # Spec 8.1: if the header is sent more than once, the first value is used.
+    keys = request.headers.getlist("Idempotency-Key")

+    request.state.total_cents = command.total_cents
+
+    result = execute_bulk_payment(session, command, settings.lock_timeout_ms)
+    request.state.outcome = "replayed" if result.replayed else "created"
+    # The original and the replay are serialized the same way, so the bytes match.
+    headers = {"Idempotent-Replayed": "true"} if result.replayed else None
+    return JSONResponse(result.body, status_code=201, headers=headers)

+import threading
+from collections.abc import Iterator

+from sqlalchemy import Engine, create_engine
+from sqlalchemy.orm import Session, sessionmaker
+
+from app.config import get_settings
+
+# One engine per process: its pool bounds the process's connections (spec section 9.3).
+# lru_cache would not do: concurrent first calls each run the wrapped function.
+_engine: Engine | None = None
+_engine_lock = threading.Lock()
+
+
+def _create_engine() -> Engine:
+    settings = get_settings()
+    return create_engine(
+        settings.database_url,
+        pool_size=settings.db_pool_size,
+        max_overflow=0,
+        pool_timeout=settings.db_pool_timeout_s,
+        pool_pre_ping=True,
+        # Error text must not carry bound parameters: they include payment
+        # descriptions, which never go to logs (spec section 12).
+        hide_parameters=True,
+    )
+
+
+def get_engine() -> Engine:
+    global _engine
+    engine = _engine
+    if engine is None:
+        with _engine_lock:
+            if _engine is None:
+                _engine = _create_engine()
+            engine = _engine
+    return engine
+
+
+def get_session() -> Iterator[Session]:
+    factory = sessionmaker(bind=get_engine(), expire_on_commit=False)
+    with factory() as session:
+        yield session

+import hashlib
+import json
+from collections.abc import Sequence
+from typing import Any, cast
+
+from sqlalchemy import select, update
+from sqlalchemy.dialects.postgresql import insert
+from sqlalchemy.orm import Session
+
+from app.errors import IdempotencyKeyReuse
+from app.models import IdempotencyKey
+
+
+def compute_request_hash(payer_uuid: str, items: Sequence[tuple[str, int, str]]) -> str:
+    """SHA-256 of the parsed request; items are (payee_uuid, amount_cents, description).
+
+    The canonical form must never change: stored hashes have to stay comparable.
+    """
+    canonical = json.dumps(
+        {"payer": payer_uuid, "payments": [list(item) for item in items]},
+        sort_keys=True,
+        separators=(",", ":"),
+        ensure_ascii=False,
+    )
+    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
+
+
+def claim_key(
+    session: Session, payer_uuid: str, key: str, request_hash: str
+) -> dict[str, Any] | None:
+    """Claim the key for this transaction, or return the response stored under it.
+
+    Returns None when the key is new. If another transaction holds an uncommitted
+    row for the key, the insert waits for its outcome (bounded by lock_timeout):
+    after a commit the stored response is returned, after a rollback the key is
+    claimed here.
+    """
+    claimed_id = session.execute(
+        # response_body is left out: SQLAlchemy's JSON type would store None as JSON null.
+        insert(IdempotencyKey)
+        .values(payer_firm_uuid=payer_uuid, key=key, request_hash=request_hash)
+        .on_conflict_do_nothing(index_elements=["payer_firm_uuid", "key"])
+        .returning(IdempotencyKey.id)
+    ).scalar_one_or_none()
+    if claimed_id is not None:
+        return None
+
+    # A new statement: under Read Committed it sees the row that won the conflict.
+    existing = session.execute(
+        select(IdempotencyKey.request_hash, IdempotencyKey.response_body).where(
+            IdempotencyKey.payer_firm_uuid == payer_uuid, IdempotencyKey.key == key
+        )
+    ).one()
+    if existing.response_body is None:
+        # The row and its response are committed together, so this cannot happen.
+        raise RuntimeError("a committed idempotency key has no stored response")
+    if existing.request_hash != request_hash:
+        raise IdempotencyKeyReuse("the Idempotency-Key was already used with a different request")
+    return cast(dict[str, Any], existing.response_body)
+
+
+def store_response(session: Session, payer_uuid: str, key: str, body: dict[str, Any]) -> None:
+    """Store the 201 body under a key this transaction has claimed."""
+    session.execute(
+        update(IdempotencyKey)
+        .where(IdempotencyKey.payer_firm_uuid == payer_uuid, IdempotencyKey.key == key)
+        .values(response_body=body)
+    )

+from contextlib import asynccontextmanager
+from typing import Any

+
+from app.api import BULK_PAYMENTS_PATH, router

+request_logger = logging.getLogger("app.requests")
+
+_HTTP_ERROR_CODES = {404: "not_found", 405: "method_not_allowed"}
+
+# Pydantic messages that quote part of the input ("found `z` at 0"), by error type.
+_FIXED_MESSAGES = {"uuid_parsing": "Input should be a valid UUID"}
+
+
+def _configure_logging() -> None:
+    # Uvicorn configures only its own loggers; without a handler INFO records are dropped.
+    app_logger = logging.getLogger("app")
+    if not app_logger.handlers:
+        handler = logging.StreamHandler()
+        handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(name)s %(message)s"))
+        app_logger.addHandler(handler)
+        app_logger.setLevel(logging.INFO)
+
+
+@asynccontextmanager
+async def lifespan(_: FastAPI) -> AsyncIterator[None]:
+    _configure_logging()
+    # Synchronous endpoints wait for a thread instead of failing on pool checkout.
+    # The limiter can be changed only from the event loop.
+    anyio.to_thread.current_default_thread_limiter().total_tokens = get_settings().db_pool_size
+    # Built before the first request, so requests never race to create it.
+    get_engine()
+    yield
+
+
+async def log_request(
+    request: Request, call_next: Callable[[Request], Awaitable[Response]]
+) -> Response:
+    """Write one record per bulk payment request; descriptions are never logged."""
+    if request.url.path != BULK_PAYMENTS_PATH:
+        return await call_next(request)
+    started = time.perf_counter()
+    request_id = request.headers.get("x-request-id") or uuid.uuid4().hex
+    status = 500
+    try:
+        response = await call_next(request)
+        status = response.status_code
+        return response
+    finally:
+        state = request.state
+        total_cents = getattr(state, "total_cents", None)
+        record = {
+            "request_id": request_id,
+            "instance": INSTANCE,
+            "payer": getattr(state, "payer", None),
+            "payments": getattr(state, "payment_count", None),
+            "total": None if total_cents is None else format_cents(total_cents),
+            # Unset only when an exception escaped to the 500 handler.
+            "outcome": getattr(state, "outcome", "internal_error"),
+            "status": status,
+            "duration_ms": round((time.perf_counter() - started) * 1000, 1),
+        }
+        request_logger.info(json.dumps(record))
+
+
+def _error_response(
+    request: Request,
+    status_code: int,
+    code: str,
+    message: str,
+    *,
+    details: list[dict[str, Any]] | None = None,
+    headers: Mapping[str, str] | None = None,
+) -> JSONResponse:
+    request.state.outcome = code
+    error: dict[str, Any] = {"code": code, "message": message}
+    if details is not None:
+        error["details"] = details
+    return JSONResponse({"error": error}, status_code=status_code, headers=headers)
+
+
+async def handle_domain_error(request: Request, exc: DomainError) -> JSONResponse:
+    details = exc.details if isinstance(exc, ValidationFailed) else None
+    headers = {"Retry-After": "1"} if isinstance(exc, ServiceBusy) else None
+    return _error_response(
+        request, exc.status_code, exc.code, exc.message, details=details, headers=headers
+    )
+
+
+async def handle_request_validation_error(
+    request: Request, exc: RequestValidationError
+) -> JSONResponse:
+    # Only loc, msg and type: input and ctx carry the rejected value, which may be
+    # sensitive or impossible to encode (a lone surrogate from a \ud800 escape).
+    details = [
+        {
+            "loc": list(error["loc"]),
+            "msg": _FIXED_MESSAGES.get(error["type"], error["msg"]),
+            "type": error["type"],
+        }
+        for error in exc.errors()
+    ]
+    return _error_response(
+        request, 422, "validation_error", "the request is invalid", details=details
+    )
+
+
+async def handle_http_exception(request: Request, exc: StarletteHTTPException) -> JSONResponse:
+    if exc.status_code == 400:
+        # FastAPI raises 400 only when it cannot read the body as JSON, for
+        # example invalid UTF-8; spec section 8.1 makes that a validation error.
+        message = "the request body is not valid JSON"
+        detail = {"loc": ["body"], "msg": message, "type": "json_invalid"}
+        return _error_response(request, 422, "validation_error", message, details=[detail])
+    code = _HTTP_ERROR_CODES.get(exc.status_code, "http_error")
+    return _error_response(request, exc.status_code, code, str(exc.detail), headers=exc.headers)
+
+
+async def handle_internal_error(request: Request, exc: Exception) -> JSONResponse:
+    # Uvicorn logs the traceback; the client learns nothing about the cause.
+    return _error_response(request, 500, "internal_error", "internal server error")
+
+
+def create_app() -> FastAPI:
+    application = FastAPI(lifespan=lifespan)
+    application.include_router(router)
+    application.middleware("http")(log_request)
+    application.exception_handler(DomainError)(handle_domain_error)
+    application.exception_handler(RequestValidationError)(handle_request_validation_error)
+    application.exception_handler(StarletteHTTPException)(handle_http_exception)
+    application.exception_handler(Exception)(handle_internal_error)
+    return application
+
+
+app = create_app()

+import re
+from typing import Annotated
+from uuid import UUID
+
+from pydantic import AfterValidator, BaseModel, ConfigDict, Field, StrictStr
+
+INT32_MAX = 2_147_483_647
+
+# 8 integer digits, a point and 2 decimals is the longest amount that fits INT32_MAX cents.
+_MAX_AMOUNT_LENGTH = 11
+# [0-9], not \d: \d also matches digits from other scripts.
+_AMOUNT_PATTERN = re.compile(r"[0-9]+(\.[0-9]{1,2})?")
+
+
+def parse_amount_cents(value: str) -> int:
+    """Convert a decimal amount string such as "5800.5" to integer cents."""
+    if len(value) > _MAX_AMOUNT_LENGTH:
+        raise ValueError(f"amount must be at most {_MAX_AMOUNT_LENGTH} characters long")
+    if _AMOUNT_PATTERN.fullmatch(value) is None:
+        raise ValueError("amount must be digits with an optional point and 1 or 2 decimals")
+    whole, _, fraction = value.partition(".")
+    cents = int(whole) * 100 + int(fraction.ljust(2, "0"))
+    if not 0 < cents <= INT32_MAX:
+        raise ValueError(f"amount must be greater than 0 and at most {format_cents(INT32_MAX)}")
+    return cents
+
+
+def format_cents(cents: int) -> str:
+    """Render non-negative integer cents as a decimal string with exactly two decimals."""
+    return f"{cents // 100}.{cents % 100:02d}"
+
+
+def _check_amount(value: str) -> str:
+    parse_amount_cents(value)
+    return value
+
+
+def _check_description(value: str) -> str:
+    # PostgreSQL TEXT cannot store U+0000.
+    if "\x00" in value:
+        raise ValueError("description must not contain the character U+0000")
+    # A lone surrogate (from a JSON escape such as \ud800) cannot be encoded
+    # for the request hash or the database.
+    try:
+        value.encode("utf-8")
+    except UnicodeEncodeError:
+        raise ValueError("description must not contain unpaired surrogates") from None
+    return value
+
+
+class PaymentIn(BaseModel):
+    model_config = ConfigDict(extra="forbid")
+
+    amount: Annotated[StrictStr, AfterValidator(_check_amount)]
+    payee_firm_uuid: UUID
+    description: Annotated[StrictStr, AfterValidator(_check_description)]
+
+
+class BulkPaymentRequest(BaseModel):
+    """The request body; limits that depend on settings are checked in build_command."""
+
+    model_config = ConfigDict(extra="forbid")
+
+    payer_firm_uuid: UUID
+    payments: Annotated[list[PaymentIn], Field(min_length=1)]

+from app.errors import (
+    BalanceOverflow,
+    FirmNotFound,
+    InsufficientFunds,
+    SelfPayment,
+    ServiceBusy,
+    ValidationFailed,
+)
+from app.idempotency import claim_key, compute_request_hash, store_response
+from app.models import Firm, Payment
+from app.schemas import INT32_MAX, BulkPaymentRequest, format_cents, parse_amount_cents
+
+logger = logging.getLogger(__name__)
+
+_LOCK_NOT_AVAILABLE = "55P03"
+_DEADLOCK_DETECTED = "40P01"
+
+# Visible ASCII, no spaces.
+_IDEMPOTENCY_KEY_PATTERN = re.compile(r"[\x21-\x7e]{1,255}")
+
+
+@dataclass(frozen=True)
+class PaymentItem:
+    payee_uuid: str
+    amount_cents: int
+    description: str
+
+
+@dataclass(frozen=True)
+class BulkPaymentCommand:
+    payer_uuid: str
+    items: tuple[PaymentItem, ...]
+    total_cents: int
+    credits: dict[str, int]
+    firm_uuids: frozenset[str]
+    idempotency_key: str | None
+    request_hash: str | None
+
+
+def _detail(loc: list[str | int], msg: str, type_: str) -> dict[str, Any]:
+    return {"loc": loc, "msg": msg, "type": type_}
+
+
+def build_command(
+    request: BulkPaymentRequest, idempotency_key: str | None, settings: Settings
+) -> BulkPaymentCommand:
+    """Check the rules the schema cannot and derive what the transaction needs.
+
+    Pure: no database access.
+    """
+    max_payments = settings.max_payments_per_request
+    if len(request.payments) > max_payments:
+        msg = f"at most {max_payments} payments are allowed per request"
+        raise ValidationFailed(msg, [_detail(["body", "payments"], msg, "too_long")])
+
+    max_length = settings.max_description_length
+    msg = f"description must be at most {max_length} characters long"
+    too_long = [
+        _detail(["body", "payments", index, "description"], msg, "string_too_long")
+        for index, payment in enumerate(request.payments)
+        if len(payment.description) > max_length
+    ]
+    if too_long:
+        raise ValidationFailed(msg, too_long)
+
+    if idempotency_key is not None and _IDEMPOTENCY_KEY_PATTERN.fullmatch(idempotency_key) is None:
+        msg = "Idempotency-Key must be 1 to 255 visible ASCII characters without spaces"
+        detail = _detail(["header", "Idempotency-Key"], msg, "string_pattern_mismatch")
+        raise ValidationFailed(msg, [detail])
+
+    payer_uuid = str(request.payer_firm_uuid)
+    items = tuple(
+        PaymentItem(
+            payee_uuid=str(payment.payee_firm_uuid),
+            amount_cents=parse_amount_cents(payment.amount),
+            description=payment.description,
+        )
+        for payment in request.payments
+    )
+
+    for index, item in enumerate(items):
+        if item.payee_uuid == payer_uuid:
+            raise SelfPayment(f"payment {index} pays the payer itself")
+
+    total_cents = sum(item.amount_cents for item in items)
+    if total_cents > INT32_MAX:
+        msg = f"the batch total must be at most {format_cents(INT32_MAX)}"
+        raise ValidationFailed(msg, [_detail(["body", "payments"], msg, "less_than_equal")])
+
+    credits: dict[str, int] = {}
+    for item in items:
+        credits[item.payee_uuid] = credits.get(item.payee_uuid, 0) + item.amount_cents
+
+    firm_uuids = frozenset({payer_uuid, *credits})
+
+    request_hash = None
+    if idempotency_key is not None:
+        request_hash = compute_request_hash(
+            payer_uuid, [(item.payee_uuid, item.amount_cents, item.description) for item in items]
+        )
+
+    return BulkPaymentCommand(
+        payer_uuid=payer_uuid,
+        items=items,
+        total_cents=total_cents,
+        credits=credits,
+        firm_uuids=firm_uuids,
+        idempotency_key=idempotency_key,
+        request_hash=request_hash,
+    )
+
+
+@dataclass(frozen=True)
+class BulkPaymentResult:
+    body: dict[str, Any]
+    replayed: bool
+
+
+def execute_bulk_payment(
+    session: Session, command: BulkPaymentCommand, lock_timeout_ms: int
+) -> BulkPaymentResult:
+    """Apply the whole batch in one transaction, or reject it and change nothing.
+
+    The session must not be in a transaction yet; this function owns it.
+    """
+    try:
+        with session.begin():
+            result = _apply(session, command, lock_timeout_ms)
+    except DBAPIError as exc:
+        sqlstate = getattr(exc.orig, "sqlstate", None)
+        if sqlstate == _DEADLOCK_DETECTED:
+            # Locks are taken in one query ordered by id, so this should not happen.
+            # exc.orig, not exc: the SQLAlchemy message carries the statement
+            # parameters, and those include payment descriptions.
+            logger.error("deadlock detected in a bulk payment transaction: %s", exc.orig)
+        if sqlstate in (_LOCK_NOT_AVAILABLE, _DEADLOCK_DETECTED):
+            raise ServiceBusy("the payment could not be processed in time; retry") from exc
+        raise
+    except PoolTimeoutError as exc:
+        raise ServiceBusy("no database connection is available; retry") from exc
+    return result
+
+
+def _apply(
+    session: Session, command: BulkPaymentCommand, lock_timeout_ms: int
+) -> BulkPaymentResult:
+    """Transaction steps of spec section 8.2; only database work happens here."""
+    session.execute(
+        text("SELECT set_config('lock_timeout', :value, true)"),
+        {"value": f"{lock_timeout_ms}ms"},
+    )
+
+    key = command.idempotency_key
+    if key is not None:
+        assert command.request_hash is not None  # build_command sets both or neither
+        # Claimed before any firm is read: a second request with this key waits
+        # here for the outcome of the first.
+        stored = claim_key(session, command.payer_uuid, key, command.request_hash)
+        if stored is not None:
+            return BulkPaymentResult(body=stored, replayed=True)
+
+    # The only read of firms: it finds, checks and locks them in one consistent order.
+    locked = session.scalars(
+        select(Firm)
+        .where(Firm.uuid.in_(command.firm_uuids))
+        .order_by(Firm.id)
+        .with_for_update(key_share=True)
+        .execution_options(populate_existing=True)
+    ).all()
+    firms = {firm.uuid: firm for firm in locked}
+
+    missing = command.firm_uuids - firms.keys()
+    if missing:
+        raise FirmNotFound(f"firms not found: {', '.join(sorted(missing))}")
+
+    payer = firms[command.payer_uuid]
+    if payer.balance_cents < command.total_cents:
+        raise InsufficientFunds("the payer's balance does not cover the batch total")
+
+    for payee_uuid, credit in command.credits.items():
+        if firms[payee_uuid].balance_cents + credit > INT32_MAX:
+            raise BalanceOverflow(f"the balance of payee {payee_uuid} would exceed the maximum")
+
+    rows = [
+        Payment(
+            payer_firm_id=payer.id,
+            payee_firm_id=firms[item.payee_uuid].id,
+            amount_cents=item.amount_cents,
+            description=item.description,
+        )
+        for item in command.items
+    ]
+    session.add_all(rows)
+    session.flush()
+
+    payer.balance_cents -= command.total_cents
+    for payee_uuid, credit in command.credits.items():
+        firms[payee_uuid].balance_cents += credit
+
+    body: dict[str, Any] = {
+        "payer_firm_uuid": command.payer_uuid,
+        "total_amount": format_cents(command.total_cents),
+        "payments": [
+            {
+                "id": row.id,
+                "payee_firm_uuid": item.payee_uuid,
+                "amount": format_cents(item.amount_cents),
+                "description": item.description,
+            }
+            for row, item in zip(rows, command.items, strict=True)
+        ],
+    }
+
+    if key is not None:
+        # Balance updates go out first, so storing the response is the last statement.
+        session.flush()
+        store_response(session, command.payer_uuid, key, body)
+    return BulkPaymentResult(body=body, replayed=False)
