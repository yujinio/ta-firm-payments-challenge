    38	def run_concurrently[T](fn: Callable[[int], T], n: int) -> list[T]:
    39	    """Run fn(0) .. fn(n - 1) on n threads released together; re-raise the first exception."""
    40	    barrier = threading.Barrier(n, timeout=THREAD_TIMEOUT_S)
    41	
    42	    def start(index: int) -> T:
    43	        barrier.wait()
    44	        return fn(index)
    45	
    46	    with ThreadPoolExecutor(max_workers=n) as executor:
    47	        futures = [executor.submit(start, index) for index in range(n)]
    48	        return [future.result(timeout=THREAD_TIMEOUT_S) for future in futures]

    97	def wait_until_blocked(db: Session, timeout_s: float = 5.0) -> None:
    98	    """Return once another backend of this database waits for a lock; fail on timeout."""
    99	    deadline = time.monotonic() + timeout_s
   100	    while True:
   101	        waiting = db.execute(
   102	            text(
   103	                "SELECT count(*) FROM pg_stat_activity"
   104	                " WHERE datname = current_database() AND pid <> pg_backend_pid()"
   105	                " AND wait_event_type = 'Lock'"
   106	            )
   107	        ).scalar_one()
   108	        # pg_stat_activity is a snapshot per transaction: end it before the next poll.
   109	        db.rollback()
   110	        if waiting:
   111	            return
   112	        if time.monotonic() > deadline:
   113	            pytest.fail(f"no backend waited for a lock within {timeout_s} s")
   114	        time.sleep(0.01)

    50	def _stop(process: subprocess.Popen[bytes]) -> None:
    51	    """SIGTERM lets Uvicorn stop its workers; SIGKILL to the whole group if it hangs."""
    52	    if process.poll() is not None:
    53	        return
    54	    process.terminate()
    55	    try:
    56	        process.wait(timeout=STOP_TIMEOUT_S)
    57	    except subprocess.TimeoutExpired:
    58	        os.killpg(process.pid, signal.SIGKILL)
    59	        process.wait()

    62	def _wait_until_ready(process: subprocess.Popen[bytes], base_url: str, log_path: Path) -> None:
    63	    """Return once /healthz answers 200 and every worker has finished its startup."""
    64	    deadline = time.monotonic() + STARTUP_TIMEOUT_S
    65	    while time.monotonic() < deadline:
    66	        if process.poll() is not None:
    67	            pytest.fail(f"the server exited with {process.returncode}:\n{log_path.read_text()}")
    68	        try:
    69	            response = httpx2.get(f"{base_url}/healthz", timeout=1.0, trust_env=False)
    70	            healthy = response.status_code == 200
    71	        except httpx2.TransportError:
    72	            healthy = False
    73	        started = log_path.read_text().count("Application startup complete.")
    74	        if healthy and started == WORKERS:
    75	            return
    76	        time.sleep(0.1)
    77	    pytest.fail(f"the server was not ready within {STARTUP_TIMEOUT_S} s:\n{log_path.read_text()}")
