.PHONY: up seed reset test lint check load

# nginx resolves the replicas only at start, and recreated replicas get new
# addresses, so the balancer is restarted after every up.
up:
	docker compose up --build -d --wait
	docker compose restart lb

seed:
	docker compose exec -T db psql -U postgres -d firm_payments < scripts/seed.sql

reset:
	docker compose exec -T db psql -U postgres -d firm_payments -v ON_ERROR_STOP=1 \
		-c "TRUNCATE payments, idempotency_keys, firms RESTART IDENTITY CASCADE" -f - < scripts/seed.sql

# The tests use the firm_payments_test database on localhost:$(DB_PORT) (tests/conftest.py).
test:
	docker compose up -d --wait db
	uv run --locked pytest

lint:
	uv run --locked ruff check .
	uv run --locked ruff format --check .
	uv run --locked mypy app

check: test lint

# Through the balancer's host port. The script adds firms with ids from 1000 and
# payments between them; `make reset` returns to the seed.
load:
	uv run --locked python scripts/load.py \
		--base-url http://localhost:$(LB_PORT) --database-url $(LOAD_DB_URL) $(ARGS)

    14	def compute_request_hash(payer_uuid: str, items: Sequence[tuple[str, int, str]]) -> str:
    15	    """SHA-256 of the parsed request; items are (payee_uuid, amount_cents, description).
    16	
    17	    The canonical form must never change: stored hashes have to stay comparable.
    18	    """
    19	    canonical = json.dumps(
    20	        {"payer": payer_uuid, "payments": [list(item) for item in items]},
    21	        sort_keys=True,
    22	        separators=(",", ":"),
    23	        ensure_ascii=False,
    24	    )
    25	    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
    26	
    27	
    28	def claim_key(
    29	    session: Session, payer_uuid: str, key: str, request_hash: str
    30	) -> dict[str, Any] | None:
    31	    """Claim the key for this transaction, or return the response stored under it.
    32	
    33	    Returns None when the key is new. If another transaction holds an uncommitted
    34	    row for the key, the insert waits for its outcome (bounded by lock_timeout):
    35	    after a commit the stored response is returned, after a rollback the key is
    36	    claimed here.
    37	    """
    38	    claimed_id = session.execute(
    39	        # response_body is left out: SQLAlchemy's JSON type would store None as JSON null.
    40	        insert(IdempotencyKey)
    41	        .values(payer_firm_uuid=payer_uuid, key=key, request_hash=request_hash)
    42	        .on_conflict_do_nothing(index_elements=["payer_firm_uuid", "key"])
    43	        .returning(IdempotencyKey.id)
    44	    ).scalar_one_or_none()
    45	    if claimed_id is not None:
    46	        return None
    47	
    48	    # A new statement: under Read Committed it sees the row that won the conflict.
    49	    existing = session.execute(
    50	        select(IdempotencyKey.request_hash, IdempotencyKey.response_body).where(
    51	            IdempotencyKey.payer_firm_uuid == payer_uuid, IdempotencyKey.key == key
    52	        )
    53	    ).one()
    54	    if existing.response_body is None:
    55	        # The row and its response are committed together, so this cannot happen.
    56	        raise RuntimeError("a committed idempotency key has no stored response")
    57	    if existing.request_hash != request_hash:
    58	        raise IdempotencyKeyReuse("the Idempotency-Key was already used with a different request")
    59	    return cast(dict[str, Any], existing.response_body)
    60	
    61	
    62	def store_response(session: Session, payer_uuid: str, key: str, body: dict[str, Any]) -> None:
    63	    """Store the 201 body under a key this transaction has claimed."""
    64	    session.execute(
    65	        update(IdempotencyKey)
    66	        .where(IdempotencyKey.payer_firm_uuid == payer_uuid, IdempotencyKey.key == key)
    67	        .values(response_body=body)
    68	    )

    55	    request: Request, call_next: Callable[[Request], Awaitable[Response]]
    56	) -> Response:
    57	    """Write one record per bulk payment request; descriptions are never logged."""
    58	    if request.url.path != BULK_PAYMENTS_PATH:

   110	
   111	    Without the header the body is not parsed either (strict_content_type).

   123	async def handle_request_validation_error(
   124	    request: Request, exc: RequestValidationError
   125	) -> JSONResponse:
   126	    # Only loc, msg and type: input and ctx carry the rejected value, which may be
   127	    # sensitive or impossible to encode (a lone surrogate from a \ud800 escape).
   128	    details = [
   129	        {
   130	            "loc": list(error["loc"]),
   131	            "msg": _FIXED_MESSAGES.get(error["type"], error["msg"]),

     9	# 8 integer digits, a point and 2 decimals is the longest amount that fits INT32_MAX cents.
    10	_MAX_AMOUNT_LENGTH = 11
    11	# [0-9], not \d: \d also matches digits from other scripts.
    12	_AMOUNT_PATTERN = re.compile(r"[0-9]+(\.[0-9]{1,2})?")

    15	def parse_amount_cents(value: str) -> int:
    16	    """Convert a decimal amount string such as "5800.5" to integer cents."""
    17	    if len(value) > _MAX_AMOUNT_LENGTH:
    18	        raise ValueError(f"amount must be at most {_MAX_AMOUNT_LENGTH} characters long")
    19	    if _AMOUNT_PATTERN.fullmatch(value) is None:
    20	        raise ValueError("amount must be digits with an optional point and 1 or 2 decimals")
    21	    whole, _, fraction = value.partition(".")
    22	    cents = int(whole) * 100 + int(fraction.ljust(2, "0"))
    23	    if not 0 < cents <= INT32_MAX:
    24	        raise ValueError(f"amount must be greater than 0 and at most {format_cents(INT32_MAX)}")
    25	    return cents

    27	
    28	def format_cents(cents: int) -> str:
    29	    """Render non-negative integer cents as a decimal string with exactly two decimals."""
    30	    return f"{cents // 100}.{cents % 100:02d}"

    33	def _check_amount(value: str) -> str:
    34	    parse_amount_cents(value)

    38	def _check_description(value: str) -> str:
    39	    # PostgreSQL TEXT cannot store U+0000.
    40	    if "\x00" in value:
    41	        raise ValueError("description must not contain the character U+0000")
    42	    # A lone surrogate (from a JSON escape such as \ud800) cannot be encoded
    43	    # for the request hash or the database.
    44	    try:
    45	        value.encode("utf-8")
    46	    except UnicodeEncodeError:
    47	        raise ValueError("description must not contain unpaired surrogates") from None
    48	    return value

    55	def build_command(
    56	    request: BulkPaymentRequest, idempotency_key: str | None, settings: Settings
    57	) -> BulkPaymentCommand:
    58	    """Check the rules the schema cannot and derive what the transaction needs.
    59	
    60	    Pure: no database access.
    61	    """
    62	    max_payments = settings.max_payments_per_request
    63	    if len(request.payments) > max_payments:
    64	        msg = f"at most {max_payments} payments are allowed per request"
    65	        raise ValidationFailed(msg, [_detail(["body", "payments"], msg, "too_long")])
    66	
    67	    max_length = settings.max_description_length
    68	    msg = f"description must be at most {max_length} characters long"
    69	    too_long = [
    70	        _detail(["body", "payments", index, "description"], msg, "string_too_long")
    71	        for index, payment in enumerate(request.payments)
    72	        if len(payment.description) > max_length
    73	    ]
    74	    if too_long:
    75	        raise ValidationFailed(msg, too_long)
    76	
    77	    if idempotency_key is not None and _IDEMPOTENCY_KEY_PATTERN.fullmatch(idempotency_key) is None:
    78	        msg = "Idempotency-Key must be 1 to 255 visible ASCII characters without spaces"
    79	        detail = _detail(["header", "Idempotency-Key"], msg, "string_pattern_mismatch")
    80	        raise ValidationFailed(msg, [detail])
    81	
    82	    payer_uuid = str(request.payer_firm_uuid)
    83	    items = tuple(
    84	        PaymentItem(
    85	            payee_uuid=str(payment.payee_firm_uuid),
    86	            amount_cents=parse_amount_cents(payment.amount),
    87	            description=payment.description,
    88	        )
    89	        for payment in request.payments
    90	    )
    91	
    92	    for index, item in enumerate(items):
    93	        if item.payee_uuid == payer_uuid:
    94	            raise SelfPayment(f"payment {index} pays the payer itself")
    95	
    96	    total_cents = sum(item.amount_cents for item in items)
    97	    if total_cents > INT32_MAX:
    98	        msg = f"the batch total must be at most {format_cents(INT32_MAX)}"
    99	        raise ValidationFailed(msg, [_detail(["body", "payments"], msg, "less_than_equal")])
   100	
   101	    credits: dict[str, int] = {}
   102	    for item in items:
   103	        credits[item.payee_uuid] = credits.get(item.payee_uuid, 0) + item.amount_cents
   104	
   105	    firm_uuids = frozenset({payer_uuid, *credits})
   106	
   107	    request_hash = None
   108	    if idempotency_key is not None:
   109	        request_hash = compute_request_hash(

   111	        )
   112	
   113	    return BulkPaymentCommand(

   115	        items=items,
   116	        total_cents=total_cents,

   130	def execute_bulk_payment(
   131	    session: Session, command: BulkPaymentCommand, lock_timeout_ms: int
   132	) -> BulkPaymentResult:
   133	    """Apply the whole batch in one transaction, or reject it and change nothing.
   134	
   135	    The session must not be in a transaction yet; this function owns it.
   136	    """
   137	    try:
   138	        with session.begin():
   139	            result = _apply(session, command, lock_timeout_ms)
   140	    except DBAPIError as exc:
   141	        sqlstate = getattr(exc.orig, "sqlstate", None)
   142	        if sqlstate == _DEADLOCK_DETECTED:
   143	            # Locks are taken in one query ordered by id, so this should not happen.
   144	            # exc.orig, not exc: the SQLAlchemy message carries the statement
   145	            # parameters, and those include payment descriptions.
   146	            logger.error("deadlock detected in a bulk payment transaction: %s", exc.orig)
   147	        if sqlstate in (_LOCK_NOT_AVAILABLE, _DEADLOCK_DETECTED):
   148	            raise ServiceBusy("the payment could not be processed in time; retry") from exc
   149	        raise
   150	    except PoolTimeoutError as exc:
   151	        raise ServiceBusy("no database connection is available; retry") from exc
   152	    return result

   155	def _apply(
   156	    session: Session, command: BulkPaymentCommand, lock_timeout_ms: int
   157	) -> BulkPaymentResult:
   158	    """Transaction steps of spec section 8.2; only database work happens here."""
   159	    session.execute(
   160	        text("SELECT set_config('lock_timeout', :value, true)"),
   161	        {"value": f"{lock_timeout_ms}ms"},
   162	    )
   163	
   164	    key = command.idempotency_key
   165	    if key is not None:
   166	        assert command.request_hash is not None  # build_command sets both or neither
   167	        # Claimed before any firm is read: a second request with this key waits
   168	        # here for the outcome of the first.
   169	        stored = claim_key(session, command.payer_uuid, key, command.request_hash)
   170	        if stored is not None:
   171	            return BulkPaymentResult(body=stored, replayed=True)
   172	
   173	    # The only read of firms: it finds, checks and locks them in one consistent order.
   174	    locked = session.scalars(
   175	        select(Firm)
   176	        .where(Firm.uuid.in_(command.firm_uuids))
   177	        .order_by(Firm.id)
   178	        .with_for_update(key_share=True)
   179	        .execution_options(populate_existing=True)
   180	    ).all()
   181	    firms = {firm.uuid: firm for firm in locked}
   182	
   183	    missing = command.firm_uuids - firms.keys()
   184	    if missing:
   185	        raise FirmNotFound(f"firms not found: {', '.join(sorted(missing))}")
   186	
   187	    payer = firms[command.payer_uuid]
   188	    if payer.balance_cents < command.total_cents:
   189	        raise InsufficientFunds("the payer's balance does not cover the batch total")
   190	
   191	    for payee_uuid, credit in command.credits.items():
   192	        if firms[payee_uuid].balance_cents + credit > INT32_MAX:
   193	            raise BalanceOverflow(f"the balance of payee {payee_uuid} would exceed the maximum")
   194	
   195	    rows = [
   196	        Payment(
   197	            payer_firm_id=payer.id,
   198	            payee_firm_id=firms[item.payee_uuid].id,
   199	            amount_cents=item.amount_cents,
   200	            description=item.description,
   201	        )
   202	        for item in command.items
   203	    ]
   204	    session.add_all(rows)
   205	    session.flush()
   206	
   207	    payer.balance_cents -= command.total_cents
   208	    for payee_uuid, credit in command.credits.items():
   209	        firms[payee_uuid].balance_cents += credit
   210	
   211	    body: dict[str, Any] = {
   212	        "payer_firm_uuid": command.payer_uuid,
   213	        "total_amount": format_cents(command.total_cents),
   214	        "payments": [
   215	            {
   216	                "id": row.id,
   217	                "payee_firm_uuid": item.payee_uuid,
   218	                "amount": format_cents(item.amount_cents),
   219	                "description": item.description,
   220	            }
   221	            for row, item in zip(rows, command.items, strict=True)
   222	        ],
   223	    }
   224	
   225	    if key is not None:
   226	        # Balance updates go out first, so storing the response is the last statement.
   227	        session.flush()
   228	        store_response(session, command.payer_uuid, key, body)
   229	    return BulkPaymentResult(body=body, replayed=False)
