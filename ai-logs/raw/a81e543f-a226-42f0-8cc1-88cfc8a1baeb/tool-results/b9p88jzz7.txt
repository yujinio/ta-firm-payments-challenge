    38	def run_concurrently[T](fn: Callable[[int], T], n: int) -> list[T]:
    39	    """Run fn(0) .. fn(n - 1) on n threads released together; re-raise the first exception."""
    40	    barrier = threading.Barrier(n, timeout=THREAD_TIMEOUT_S)
    41	
    42	    def start(index: int) -> T:
    43	        """Wait until all n threads are ready, then run fn(index)."""
    44	        barrier.wait()
    45	        return fn(index)
    46	
    47	    with ThreadPoolExecutor(max_workers=n) as executor:
    48	        futures = [executor.submit(start, index) for index in range(n)]
    49	        return [future.result(timeout=THREAD_TIMEOUT_S) for future in futures]

    99	def wait_until_blocked(db: Session, timeout_s: float = 5.0) -> None:
   100	    """Return once another backend of this database waits for a lock; fail on timeout."""
   101	    deadline = time.monotonic() + timeout_s
   102	    while True:
   103	        waiting = db.execute(
   104	            text(
   105	                "SELECT count(*) FROM pg_stat_activity"
   106	                " WHERE datname = current_database() AND pid <> pg_backend_pid()"
   107	                " AND wait_event_type = 'Lock'"
   108	            )
   109	        ).scalar_one()
   110	        # pg_stat_activity is a snapshot per transaction: end it before the next poll.
   111	        db.rollback()
   112	        if waiting:
   113	            return
   114	        if time.monotonic() > deadline:
   115	            pytest.fail(f"no backend waited for a lock within {timeout_s} s")
   116	        time.sleep(0.01)

    45	    """A loopback port that was free a moment ago; the server binds it right after."""
    46	    with socket.socket() as sock:
    47	        sock.bind(("127.0.0.1", 0))
    48	        return int(sock.getsockname()[1])

    51	def _stop(process: subprocess.Popen[bytes]) -> None:
    52	    """SIGTERM lets Uvicorn stop its workers; SIGKILL to the whole group if it hangs."""
    53	    if process.poll() is not None:
    54	        return
    55	    process.terminate()
    56	    try:
    57	        process.wait(timeout=STOP_TIMEOUT_S)
    58	    except subprocess.TimeoutExpired:
    59	        os.killpg(process.pid, signal.SIGKILL)
    60	        process.wait()

    63	def _wait_until_ready(process: subprocess.Popen[bytes], base_url: str, log_path: Path) -> None:
    64	    """Return once /healthz answers 200 and every worker has finished its startup."""
    65	    deadline = time.monotonic() + STARTUP_TIMEOUT_S
    66	    while time.monotonic() < deadline:
    67	        if process.poll() is not None:
    68	            pytest.fail(f"the server exited with {process.returncode}:\n{log_path.read_text()}")
    69	        try:
    70	            response = httpx2.get(f"{base_url}/healthz", timeout=1.0, trust_env=False)
    71	            healthy = response.status_code == 200
    72	        except httpx2.TransportError:
    73	            healthy = False
    74	        started = log_path.read_text().count("Application startup complete.")
    75	        if healthy and started == WORKERS:
    76	            return
    77	        time.sleep(0.1)
    78	    pytest.fail(f"the server was not ready within {STARTUP_TIMEOUT_S} s:\n{log_path.read_text()}")

   108	@contextmanager
   109	def http_clients(base_url: str, n: int) -> Iterator[list[httpx2.Client]]:
   110	    """One client per thread, created before the threads start so they post together."""
   111	    with ExitStack() as stack:
   112	        yield [
   113	            stack.enter_context(
   114	                httpx2.Client(base_url=base_url, timeout=CLIENT_TIMEOUT_S, trust_env=False)
   115	            )
   116	            for _ in range(n)
   117	        ]

    82	def _c3_batches(firms: list[str]) -> list[dict]:
    83	    rng = random.Random(20261121)
    84	    a, b, c = firms[:3]
    85	    cycle = [bulk(a, (b, "100")), bulk(b, (c, "100")), bulk(c, (a, "100"))]
    86	    batches = []
    87	    for index in range(200):
    88	        if index % 10 < 3:
    89	            # Three legs of the cycle in every ten batches; with batch i on thread
    90	            # i % 16, the legs at 0-2 and 10-12 start together at the barrier.
    91	            batches.append(cycle[index % 10])
    92	            continue
    93	        payer = rng.choice(firms)
    94	        others = [firm for firm in firms if firm != payer]
    95	        payments = [
    96	            (rng.choice(others), _amount(rng.randint(1, 20_000))) for _ in range(rng.randint(1, 4))
    97	        ]
    98	        batches.append(bulk(payer, *payments))
    99	    return batches
