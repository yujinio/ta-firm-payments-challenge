    11	# nginx resolves the replicas only at start, and recreated replicas get new

    14		docker compose up --build -d --wait
    15		docker compose restart lb
    16

    18		docker compose exec -T db psql -U postgres -d firm_payments < scripts/seed.sql
    19	
    20	reset:
    21		docker compose exec -T db psql -U postgres -d firm_payments -v ON_ERROR_STOP=1 \
    22			-c "TRUNCATE payments, idempotency_keys, firms RESTART IDENTITY CASCADE" -f - < scripts/seed.sql
    23	
    24	# The tests use the firm_payments_test database on localhost:$(DB_PORT) (tests/conftest.py).
    25	test:
    26		docker compose up -d --wait db
    27		uv run --locked pytest
    28	
    29	lint:
    30		uv run --locked ruff check .
    31		uv run --locked ruff format --check .
    32		uv run --locked mypy app scripts migrations
    33	
    34	check: test lint
    35	
    36	# Through the balancer's host port. The script adds firms with ids from 1000 and
    37	# payments between them; `make reset` returns to the seed.
    38	load:
    39		uv run --locked python scripts/load.py \
    40			--base-url http://localhost:$(LB_PORT) --database-url $(LOAD_DB_URL) $(ARGS)
