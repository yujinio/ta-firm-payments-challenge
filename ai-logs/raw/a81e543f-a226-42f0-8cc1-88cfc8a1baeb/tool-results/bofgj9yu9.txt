    15	def compute_request_hash(payer_uuid: str, items: Sequence[tuple[str, int, str]]) -> str:
    16	    """SHA-256 of the parsed request; items are (payee_uuid, amount_cents, description).
    17	
    18	    The canonical form must never change: stored hashes have to stay comparable.
    19	    """
    20	    canonical = json.dumps(
    21	        {"payer": payer_uuid, "payments": [list(item) for item in items]},
    22	        sort_keys=True,
    23	        separators=(",", ":"),
    24	        ensure_ascii=False,
    25	    )
    26	    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

    29	def claim_key(
    30	    session: Session, payer_uuid: str, key: str, request_hash: str
    31	) -> BulkPaymentResponse | None:
    32	    """Claim the key for this transaction, or return the response stored under it.
    33	
    34	    Returns None when the key is new. If another transaction holds an uncommitted
    35	    row for the key, the insert waits for its outcome (bounded by lock_timeout):
    36	    after a commit the stored response is returned, after a rollback the key is
    37	    claimed here.
    38	    """
    39	    claimed_id = session.execute(
    40	        # response_body is left out: SQLAlchemy's JSON type would store None as JSON null.
    41	        insert(IdempotencyKey)
    42	        .values(payer_firm_uuid=payer_uuid, key=key, request_hash=request_hash)
    43	        .on_conflict_do_nothing(index_elements=["payer_firm_uuid", "key"])
    44	        .returning(IdempotencyKey.id)
    45	    ).scalar_one_or_none()
    46	    if claimed_id is not None:
    47	        return None
    48	
    49	    # A new statement: under Read Committed it sees the row that won the conflict.
    50	    existing = session.execute(
    51	        select(IdempotencyKey.request_hash, IdempotencyKey.response_body).where(
    52	            IdempotencyKey.payer_firm_uuid == payer_uuid, IdempotencyKey.key == key
    53	        )
    54	    ).one()
    55	    if existing.response_body is None:
    56	        # The row and its response are committed together, so this cannot happen.
    57	        raise RuntimeError("a committed idempotency key has no stored response")
    58	    if existing.request_hash != request_hash:
    59	        raise IdempotencyKeyReuse("the Idempotency-Key was already used with a different request")
    60	    # Not checked at runtime: only store_response writes the column, with this shape.
    61	    return cast(BulkPaymentResponse, existing.response_body)
    62	
    63	
    64	def store_response(session: Session, payer_uuid: str, key: str, body: BulkPaymentResponse) -> None:
    65	    """Store the 201 body under a key this transaction has claimed."""
    66	    session.execute(
    67	        update(IdempotencyKey)
    68	        .where(IdempotencyKey.payer_firm_uuid == payer_uuid, IdempotencyKey.key == key)
    69	        .values(response_body=body)
    70	    )

    86	async def log_request(
    87	    request: Request, call_next: Callable[[Request], Awaitable[Response]]
    88	) -> Response:
    89	    """Write one record per bulk payment request; descriptions are never logged."""
    90	    if request.url.path != BULK_PAYMENTS_PATH:
    91	        return await call_next(request)
    92	    started = time.perf_counter()
    93	    request_id = request.headers.get("x-request-id") or uuid.uuid4().hex
    94	    status = 500
    95	    try:
    96	        response = await call_next(request)
    97	        status = response.status_code
    98	        return response
    99	    finally:
   100	        state = request.state
   101	        # request.state is untyped: these getattr values are Any at this boundary.
   102	        total_cents = getattr(state, "total_cents", None)
   103	        record: RequestLogRecord = {
   104	            "request_id": request_id,
   105	            "instance": INSTANCE,
   106	            "payer": getattr(state, "payer", None),
   107	            "payments": getattr(state, "payment_count", None),
   108	            "total": None if total_cents is None else format_cents(total_cents),
   109	            # Unset only when an exception escaped to the 500 handler.
   110	            "outcome": getattr(state, "outcome", "internal_error"),
   111	            "status": status,
   112	            "duration_ms": round((time.perf_counter() - started) * 1000, 1),
   113	        }
   114	        request_logger.info(json.dumps(record))

   159	async def handle_request_validation_error(
   160	    request: Request, exc: RequestValidationError
   161	) -> JSONResponse:
   162	    """Answer a request FastAPI rejected before the endpoint ran with 422 validation_error."""
   163	    # Only loc, msg and type: input and ctx carry the rejected value, which may be
   164	    # sensitive or impossible to encode (a lone surrogate from a \ud800 escape).
   165	    # exc.errors() is Sequence[Any]: FastAPI does not type Pydantic's raw error dicts.
   166	    details: list[ErrorDetail] = [
   167	        {
   168	            "loc": list(error["loc"]),
   169	            "msg": _FIXED_MESSAGES.get(error["type"], error["msg"]),
   170	            "type": error["type"],
   171	        }
   172	        for error in exc.errors()
   173	    ]
   174	    if _is_json_content_type(request.headers.get("content-type")):
   175	        message = "the request is invalid"
   176	    else:
   177	        # The body reached validation as raw bytes, for example from `curl -d`.
   178	        message = "the request body must be JSON sent with Content-Type: application/json"
   179	    return _error_response(request, 422, "validation_error", message, details=details)

   182	async def handle_http_exception(request: Request, exc: StarletteHTTPException) -> JSONResponse:
   183	    """Answer an HTTP error raised by the framework in the service's error format.
   184	
   185	    A 400, a body that cannot be read as JSON, becomes 422 validation_error.
   186	    """
   187	    if exc.status_code == 400:
   188	        # FastAPI raises 400 only when it cannot read the body as JSON, for
   189	        # example invalid UTF-8; spec section 8.1 makes that a validation error.
   190	        message = "the request body is not valid JSON"
   191	        detail: ErrorDetail = {"loc": ["body"], "msg": message, "type": "json_invalid"}
   192	        return _error_response(request, 422, "validation_error", message, details=[detail])
   193	    code = _HTTP_ERROR_CODES.get(exc.status_code, "http_error")
   194	    return _error_response(request, exc.status_code, code, str(exc.detail), headers=exc.headers)

    10	_MAX_AMOUNT_LENGTH = 11

    15	def parse_amount_cents(value: str) -> int:
    16	    """Convert a decimal amount string such as "5800.5" to integer cents."""
    17	    if len(value) > _MAX_AMOUNT_LENGTH:
    18	        raise ValueError(f"amount must be at most {_MAX_AMOUNT_LENGTH} characters long")
    19	    if _AMOUNT_PATTERN.fullmatch(value) is None:
    20	        raise ValueError("amount must be digits with an optional point and 1 or 2 decimals")
    21	    whole, _, fraction = value.partition(".")
    22	    cents = int(whole) * 100 + int(fraction.ljust(2, "0"))
    23	    if not 0 < cents <= INT32_MAX:
    24	        raise ValueError(f"amount must be greater than 0 and at most {format_cents(INT32_MAX)}")
    25	    return cents

    71	    request: BulkPaymentRequest, idempotency_key: str | None, settings: Settings
    72	) -> BulkPaymentCommand:

    74	
    75	    Pure: no database access. Raises ValidationFailed or SelfPayment.
    76	    """
    77	    max_payments = settings.max_payments_per_request
    78	    if len(request.payments) > max_payments:
    79	        msg = f"at most {max_payments} payments are allowed per request"
    80	        raise ValidationFailed(msg, [_detail(["body", "payments"], msg, "too_long")])

    82	    max_length = settings.max_description_length
    83	    msg = f"description must be at most {max_length} characters long"
    84	    too_long = [
    85	        _detail(["body", "payments", index, "description"], msg, "string_too_long")
    86	        for index, payment in enumerate(request.payments)

    92	    if idempotency_key is not None and _IDEMPOTENCY_KEY_PATTERN.fullmatch(idempotency_key) is None:

    94	        detail = _detail(["header", "Idempotency-Key"], msg, "string_pattern_mismatch")

   147	def execute_bulk_payment(
   148	    session: Session, command: BulkPaymentCommand, lock_timeout_ms: int
   149	) -> BulkPaymentResult:
   150	    """Apply the whole batch in one transaction, or reject it and change nothing.
   151	
   152	    The session must not be in a transaction yet; this function owns it. A rejection
   153	    raises a DomainError: FirmNotFound, InsufficientFunds, BalanceOverflow,
   154	    IdempotencyKeyReuse, or ServiceBusy on a lock timeout, a deadlock or a pool
   155	    timeout. Other database errors propagate.
   156	    """
   157	    try:
   158	        with session.begin():
   159	            result = _apply(session, command, lock_timeout_ms)
   160	    except DBAPIError as exc:
   161	        sqlstate = getattr(exc.orig, "sqlstate", None)
   162	        if sqlstate == _DEADLOCK_DETECTED:
   163	            # Locks are taken in one query ordered by id, so this should not happen.
   164	            # exc.orig, not exc: the SQLAlchemy message carries the statement
   165	            # parameters, and those include payment descriptions.
   166	            logger.error("deadlock detected in a bulk payment transaction: %s", exc.orig)
   167	        if sqlstate in (_LOCK_NOT_AVAILABLE, _DEADLOCK_DETECTED):
   168	            raise ServiceBusy("the payment could not be processed in time; retry") from exc
   169	        raise
   170	    except PoolTimeoutError as exc:
   171	        raise ServiceBusy("no database connection is available; retry") from exc
   172	    return result
   173	
   174	
   175	def _apply(
   176	    session: Session, command: BulkPaymentCommand, lock_timeout_ms: int
   177	) -> BulkPaymentResult:
   178	    """Transaction steps of spec section 8.2; only database work happens here."""
   179	    session.execute(
   180	        text("SELECT set_config('lock_timeout', :value, true)"),
   181	        {"value": f"{lock_timeout_ms}ms"},
   182	    )
   183	
   184	    key = command.idempotency_key
   185	    if key is not None:
   186	        assert command.request_hash is not None  # build_command sets both or neither
   187	        # Claimed before any firm is read: a second request with this key waits
   188	        # here for the outcome of the first.
   189	        stored = claim_key(session, command.payer_uuid, key, command.request_hash)
   190	        if stored is not None:
   191	            return BulkPaymentResult(body=stored, replayed=True)
   192	
   193	    # The only read of firms: it finds, checks and locks them in one consistent order.
   194	    locked = session.scalars(
   195	        select(Firm)
   196	        .where(Firm.uuid.in_(command.firm_uuids))
   197	        .order_by(Firm.id)
   198	        .with_for_update(key_share=True)
   199	        .execution_options(populate_existing=True)
   200	    ).all()
   201	    firms = {firm.uuid: firm for firm in locked}
   202	
   203	    missing = command.firm_uuids - firms.keys()
   204	    if missing:
   205	        raise FirmNotFound(f"firms not found: {', '.join(sorted(missing))}")
   206	
   207	    payer = firms[command.payer_uuid]
   208	    if payer.balance_cents < command.total_cents:
   209	        raise InsufficientFunds("the payer's balance does not cover the batch total")
   210	
   211	    for payee_uuid, credit in command.credits.items():
   212	        if firms[payee_uuid].balance_cents + credit > INT32_MAX:
   213	            raise BalanceOverflow(f"the balance of payee {payee_uuid} would exceed the maximum")
   214	
   215	    rows = [
   216	        Payment(
   217	            payer_firm_id=payer.id,
   218	            payee_firm_id=firms[item.payee_uuid].id,
   219	            amount_cents=item.amount_cents,
   220	            description=item.description,
   221	        )
   222	        for item in command.items
   223	    ]
   224	    session.add_all(rows)
   225	    session.flush()
   226	
   227	    payer.balance_cents -= command.total_cents
   228	    for payee_uuid, credit in command.credits.items():
   229	        firms[payee_uuid].balance_cents += credit
   230	
   231	    body: BulkPaymentResponse = {
   232	        "payer_firm_uuid": command.payer_uuid,
   233	        "total_amount": format_cents(command.total_cents),
   234	        "payments": [
   235	            {
   236	                "id": row.id,
   237	                "payee_firm_uuid": item.payee_uuid,
   238	                "amount": format_cents(item.amount_cents),
   239	                "description": item.description,
   240	            }
   241	            for row, item in zip(rows, command.items, strict=True)
   242	        ],
   243	    }
   244	
   245	    if key is not None:
   246	        # Balance updates go out first, so storing the response is the last statement.
   247	        session.flush()
   248	        store_response(session, command.payer_uuid, key, body)
   249	    return BulkPaymentResult(body=body, replayed=False)
